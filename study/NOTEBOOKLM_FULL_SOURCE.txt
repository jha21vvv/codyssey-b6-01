-- ==============================================================================
-- 스마트 카페 주문 시스템 (Smart Cafe Ordering Database)
-- 파일명: sql/01_schema.sql
-- 설명: 데이터베이스 테이블 생성(DDL) 및 무결성 제약조건 정의 스크립트
-- ==============================================================================

-- [1차]: SQLite 엔진에서 외래키(Foreign Key) 참조 무결성 검사를 강제로 활성화합니다.
-- [2차]: 자물쇠를 채워두지 않으면 엉뚱한 부모 번호를 가진 고아 데이터가 몰래 들어올 수 있으므로 현관문 보안 시스템을 켭니다.
PRAGMA foreign_keys = ON;

-- [1차]: 기존에 동일한 이름의 자식 테이블(order_items)이 존재할 경우 삭제합니다.
-- [2차]: 리모델링 공사 전, 부모 테이블을 지우려면 먼저 자식 테이블 방부터 깨끗이 비워야 합니다.
DROP TABLE IF EXISTS order_items;

-- [1차]: 기존에 동일한 이름의 주문 테이블(orders)이 존재할 경우 삭제합니다.
-- [2차]: 주문 상세 방을 비운 다음, 주문서 책상을 안전하게 치웁니다.
DROP TABLE IF EXISTS orders;

-- [1차]: 기존에 동일한 이름의 메뉴 테이블(menus)이 존재할 경우 삭제합니다.
-- [2차]: 카테고리 간판을 내리기 전에, 메뉴판 메뉴 목록을 먼저 안전하게 정리합니다.
DROP TABLE IF EXISTS menus;

-- [1차]: 기존에 동일한 이름의 카테고리 테이블(categories)이 존재할 경우 삭제합니다.
-- [2차]: 메뉴들이 모두 치워졌으므로 이제 카테고리 분류 서랍장을 철거합니다.
DROP TABLE IF EXISTS categories;

-- [1차]: 기존에 동일한 이름의 회원 테이블(members)이 존재할 경우 삭제합니다.
-- [2차]: 주문 내역이 모두 정리되었으므로 이제 회원 명부 서랍장을 철거합니다.
DROP TABLE IF EXISTS members;

-- ==============================================================================
-- 1. members 테이블: 고객 및 회원 정보 관리
-- ==============================================================================
-- [1차]: 회원(members) 정보를 저장하는 독립 테이블을 생성합니다.
-- [2차]: 카페 단골 고객들의 명함을 차곡차곡 모아둘 새 회원 수첩을 만듭니다.
CREATE TABLE members (
    -- [1차]: 정수형 기본키(PK)로 자동 증가(AUTOINCREMENT) 속성을 부여합니다.
    -- [2차]: 손님이 올 때마다 1번, 2번, 3번처럼 자동으로 번호표를 발급해 주는 고객 고유 식별 번호입니다.
    member_id INTEGER PRIMARY KEY AUTOINCREMENT,

    -- [1차]: 가변 문자열이며 중복이 불가능(UNIQUE)하고 빈 값을 허용하지 않습니다(NOT NULL).
    -- [2차]: 로그인 아이디 역할을 하는 이메일은 절대 비워둘 수 없고, 다른 사람과 겹쳐서도 안 되는 주민번호 같은 존재입니다.
    email TEXT NOT NULL UNIQUE,

    -- [1차]: 고객의 실명을 저장하며 빈 값을 허용하지 않습니다(NOT NULL).
    -- [2차]: 음료가 나왔을 때 불러드릴 손님의 이름이므로 절대 비워둘 수 없습니다.
    name TEXT NOT NULL,

    -- [1차]: 고객 연락처를 저장하며 빈 값을 허용하지 않습니다(NOT NULL).
    -- [2차]: 주문 완료 알림 문자를 전송해야 하므로 전화번호는 필수 입력 사항입니다.
    phone TEXT NOT NULL,

    -- [1차]: 고객 등급을 저장하며 지정된 세 가지 문자열 중 하나만 허용(CHECK 제약조건)하고 기본값은 'NORMAL'입니다.
    -- [2차]: 손님 등급은 오직 일반(NORMAL), 우수(VIP), 최우수(VVIP) 3개 등급만 가능하며 처음 오신 분은 일반 등급 명찰을 받습니다.
    grade TEXT NOT NULL DEFAULT 'NORMAL' CHECK(grade IN ('NORMAL', 'VIP', 'VVIP')),

    -- [1차]: 누적 포인트를 저장하는 정수형 컬럼으로 음수 방지 CHECK 및 기본값 0을 적용합니다.
    -- [2차]: 카페 스탬프 포인트는 0점부터 시작하며, 포인트 통장이 마이너스가 되는 황당한 상황을 방지합니다.
    points INTEGER NOT NULL DEFAULT 0 CHECK(points >= 0),

    -- [1차]: 가입 일시를 기록하며 기본값으로 레코드 생성 시점의 현재 시각을 자동 저장합니다.
    -- [2차]: 회원이 가입 도장을 찍는 순간의 날짜와 시간을 시스템 시계가 자동으로 기록합니다.
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ==============================================================================
-- 2. categories 테이블: 음료 및 디저트 메뉴 대분류
-- ==============================================================================
-- [1차]: 메뉴의 상위 분류(커피, 에이드, 베이커리 등)를 정의하는 카테고리 테이블을 생성합니다.
-- [2차]: 카페 메뉴판의 큰 섹션 제목(커피류, 차류, 빵류)을 분류해 둘 카테고리 서랍장을 만듭니다.
CREATE TABLE categories (
    -- [1차]: 카테고리 식별용 정수형 기본키입니다.
    -- [2차]: 각 메뉴 분류 섹션에 붙이는 고유 번호표(1: 커피, 2: 논커피 등)입니다.
    category_id INTEGER PRIMARY KEY AUTOINCREMENT,

    -- [1차]: 카테고리 이름으로 고유해야 하며 빈 값을 허용하지 않습니다.
    -- [2차]: '커피'라는 카테고리가 메뉴판에 두 번 중복해서 생기면 손님이 혼란스러우므로 유일해야 합니다.
    category_name TEXT NOT NULL UNIQUE,

    -- [1차]: 카테고리에 대한 추가 설명 텍스트 컬럼입니다.
    -- [2차]: 손님에게 해당 카테고리가 어떤 음료들인지 소개해 주는 친절한 안내 문구입니다.
    description TEXT
);

-- ==============================================================================
-- 3. menus 테이블: 판매 음료 및 푸드 상세 메뉴
-- ==============================================================================
-- [1차]: 개별 메뉴 상품의 정보를 관리하는 테이블을 생성합니다.
-- [2차]: 카페에서 판매하는 실제 음료와 디저트 하나하나의 상세 정보가 적힌 개별 메뉴 카드들을 모읍니다.
CREATE TABLE menus (
    -- [1차]: 메뉴 식별용 정수형 기본키입니다.
    -- [2차]: '아메리카노', '카페라떼' 등 각 메뉴 상품에 매겨진 고유 바코드 번호입니다.
    menu_id INTEGER PRIMARY KEY AUTOINCREMENT,

    -- [1차]: categories 테이블의 category_id를 참조하는 외래키(FK) 컬럼입니다. (1:N 관계)
    -- [2차]: 이 메뉴가 메뉴판의 어떤 분류 섹션(부모 서랍)에 속해 있는지 가리키는 꼬리표입니다.
    category_id INTEGER NOT NULL,

    -- [1차]: 메뉴 명칭을 저장하며 필수 입력입니다.
    -- [2차]: 상품의 이름이 없으면 주문을 받을 수 없으므로 이름은 필수입니다.
    menu_name TEXT NOT NULL,

    -- [1차]: 판매 정가를 저장하며 0보다 커야 하는 CHECK 제약조건을 부여합니다.
    -- [2차]: 커피 가격이 0원이거나 마이너스면 가게가 망하므로 무조건 0원 초과 금액만 허용합니다.
    price INTEGER NOT NULL CHECK(price > 0),

    -- [1차]: 품절 여부를 나타내는 불리언(0 또는 1) 플래그 컬럼입니다. 기본값 1(판매중).
    -- [2차]: 재료가 떨어졌을 때 메뉴판에 '품절' 딱지를 붙이거나 떼는 스위치(1은 판매중, 0은 품절)입니다.
    is_available INTEGER NOT NULL DEFAULT 1 CHECK(is_available IN (0, 1)),

    -- [1차]: 메뉴 등록 일시를 저장합니다.
    -- [2차]: 신메뉴가 출시되어 메뉴판에 처음 등록된 날짜를 기록합니다.
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,

    -- [1차]: 외래키 제약조건: categories의 category_id를 참조하며 부모 카테고리 삭제 시 차단(RESTRICT)합니다.
    -- [2차]: 메뉴가 남아있는데 부모 카테고리를 갑자기 없애버리면 고아 메뉴가 되므로 함부로 지우지 못하게 막습니다.
    FOREIGN KEY (category_id) REFERENCES categories (category_id) ON DELETE RESTRICT
);

-- ==============================================================================
-- 4. orders 테이블: 고객의 주문 영수증 헤더
-- ==============================================================================
-- [1차]: 고객의 주문 기본 정보를 관리하는 마스터 주문 테이블을 생성합니다.
-- [2차]: 고객이 카운터나 키오스크에서 결제할 때 한 번에 발행되는 주문 영수증의 겉표지입니다.
CREATE TABLE orders (
    -- [1차]: 주문 식별용 정수형 기본키입니다.
    -- [2차]: 영수증 상단에 인쇄되는 세상에 단 하나뿐인 주문 영수증 번호입니다.
    order_id INTEGER PRIMARY KEY AUTOINCREMENT,

    -- [1차]: members 테이블의 member_id를 참조하는 외래키(FK) 컬럼입니다. (1:N 관계)
    -- [2차]: 이 영수증이 어떤 단골 회원 손님이 주문한 것인지 가리키는 손님 회원번호입니다.
    member_id INTEGER NOT NULL,

    -- [1차]: 주문 상태를 정의하며 지정된 4개 상태 중 하나만 허용합니다.
    -- [2차]: 주문 접수(PENDING), 결제완료(PAID), 취소(CANCELLED), 픽업완료(COMPLETED) 상태를 추적합니다.
    order_status TEXT NOT NULL DEFAULT 'PAID' CHECK(order_status IN ('PENDING', 'PAID', 'CANCELLED', 'COMPLETED')),

    -- [1차]: 해당 주문의 총 결제 금액을 기록합니다.
    -- [2차]: 이 주문 영수증에 적힌 모든 음료 가격을 다 더한 최종 영수증 총액입니다.
    total_amount INTEGER NOT NULL DEFAULT 0 CHECK(total_amount >= 0),

    -- [1차]: 결제 수단(CARD, CASH, POINT, KAKAO_PAY)을 저장합니다.
    -- [2차]: 손님이 신용카드, 현금, 포인트, 카카오페이 중 무엇으로 돈을 냈는지 표시합니다.
    payment_method TEXT NOT NULL CHECK(payment_method IN ('CARD', 'CASH', 'POINT', 'KAKAO_PAY')),

    -- [1차]: 주문이 발생한 일시를 기록합니다.
    -- [2차]: 카운터에서 포스기 결제 버튼을 누른 정확한 시각입니다.
    ordered_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,

    -- [1차]: 외래키 제약: 회원이 삭제되려고 할 때 기존 주문 내역이 있으면 삭제를 방지(RESTRICT)합니다.
    -- [2차]: 과거 결제 영수증이 남아있는 손님은 회계 감사를 위해 회원 명부에서 함부로 없앨 수 없습니다.
    FOREIGN KEY (member_id) REFERENCES members (member_id) ON DELETE RESTRICT
);

-- ==============================================================================
-- 5. order_items 테이블: 주문 영수증 내의 세부 주문 품목 (N:M 해소 중간 테이블)
-- ==============================================================================
-- [1차]: 한 번의 주문에 담긴 개별 메뉴와 수량 정보를 보관하는 상세 테이블을 생성합니다.
-- [2차]: 영수증 안쪽에 적힌 "아메리카노 2잔, 치즈케이크 1개"처럼 줄줄이 적히는 상세 구매 품목 목록입니다.
CREATE TABLE order_items (
    -- [1차]: 주문 상세 품목 식별용 정수형 기본키입니다.
    -- [2차]: 영수증 속 세부 품목 한 줄 한 줄마다 매겨지는 고유 일련번호입니다.
    order_item_id INTEGER PRIMARY KEY AUTOINCREMENT,

    -- [1차]: orders 테이블의 order_id를 참조하는 외래키(FK) 컬럼입니다. (1:N 관계)
    -- [2차]: 이 품목 줄이 몇 번 주문 영수증에 적혀 있는 것인지 묶어주는 영수증 번호입니다.
    order_id INTEGER NOT NULL,

    -- [1차]: menus 테이블의 menu_id를 참조하는 외래키(FK) 컬럼입니다. (1:N 관계)
    -- [2차]: 주문한 메뉴가 메뉴판의 어떤 메뉴인지 가리키는 메뉴 식별 바코드 번호입니다.
    menu_id INTEGER NOT NULL,

    -- [1차]: 주문한 메뉴의 수량으로 1개 이상이어야 합니다.
    -- [2차]: 음료를 몇 잔 시켰는지 나타내며, 0잔이나 마이너스 잔 주문은 있을 수 없으므로 1 이상이어야 합니다.
    quantity INTEGER NOT NULL CHECK(quantity > 0),

    -- [1차]: 주문 발생 시점의 개별 메뉴 단가를 저장합니다.
    -- [2차]: 훗날 커피 정가가 5,000원으로 올라도 옛날 영수증에는 당시 결제한 4,500원이 그대로 박제되어 있어야 합니다.
    unit_price INTEGER NOT NULL CHECK(unit_price > 0),

    -- [1차]: 외래키 제약: 부모 주문(orders) 삭제 시 상세 품목들도 함께 연쇄 삭제(CASCADE)합니다.
    -- [2차]: 영수증 원본을 파쇄하면 영수증 안의 세부 구매 품목 내역도 함께 휴지통으로 버려집니다.
    FOREIGN KEY (order_id) REFERENCES orders (order_id) ON DELETE CASCADE,

    -- [1차]: 외래키 제약: 메뉴가 과거 주문 상세에 물려있다면 메뉴 삭제를 차단(RESTRICT)합니다.
    -- [2차]: 과거 영수증에 버젓이 찍힌 메뉴를 메뉴판 DB에서 완전히 삭제해버리면 영수증이 훼손되므로 막습니다.
    FOREIGN KEY (menu_id) REFERENCES menus (menu_id) ON DELETE RESTRICT
);

-- ==============================================================================
-- 스마트 카페 주문 시스템 (Smart Cafe Ordering Database)
-- 파일명: sql/02_sample_data.sql
-- 설명: 각 테이블당 10행 이상의 현실감 있는 샘플 데이터 입력(DML) 스크립트
-- 주의: 참조 무결성(FK)을 준수하기 위해 부모 테이블부터 순서대로 데이터를 삽입합니다.
-- ==============================================================================

-- [1차]: 외래키 무결성 제약조건을 엄격히 활성화합니다.
-- [2차]: 부모 없는 미아 데이터가 끼어들지 못하도록 현관문 경비원에게 출입증 검사를 지시합니다.
PRAGMA foreign_keys = ON;

-- ==============================================================================
-- 1. members 테이블 샘플 데이터 (10명 이상)
-- ==============================================================================
-- [1차]: 신규 회원 10명의 정보를 members 테이블에 일괄 삽입합니다.
-- [2차]: 새로 개업한 카페에 처음 방문한 10명의 단골 손님 명단을 회원 수첩에 차례대로 기록합니다.
INSERT INTO members (member_id, email, name, phone, grade, points, created_at) VALUES

-- [1차]: SQLite 세션에서 외래키 제약조건을 강제하여 무결성을 유지합니다.

PRAGMA foreign_keys = ON;

-- ------------------------------------------------------------------------------
-- Q01. 판매 중인 메뉴 목록을 가격이 비싼 순서로 상위 5개 조회
-- [쿼리 목적]: 현재 메뉴판에 활성화(판매중)된 프리미엄 메뉴 TOP 5를 확인하여 주력 상품 파악
-- ------------------------------------------------------------------------------
-- [1차]: 메뉴 ID, 이름, 가격 컬럼을 프로젝션(SELECT)합니다.
-- [2차]: 메뉴판에서 손님에게 보여줄 '메뉴 번호, 이름, 가격표' 3가지만 골라냅니다.
SELECT menu_id, menu_name, price
-- [1차]: menus 테이블에서 데이터를 가져옵니다.
-- [2차]: 카페 전체 메뉴 카드가 꽂혀 있는 메뉴판 서랍을 엽니다.
FROM menus
-- [1차]: 판매 가능(is_available = 1) 플래그를 가진 레코드만 필터링합니다.
-- [2차]: 현재 품절 딱지가 붙지 않고 정상 판매 중인 메뉴들만 쏙쏙 추려냅니다.
WHERE is_available = 1
-- [1차]: 가격(price)을 기준으로 내림차순(DESC) 정렬합니다.
-- [2차]: 가장 비싸고 고급스러운 프리미엄 메뉴부터 맨 위에 오도록 줄을 세웁니다.
ORDER BY price DESC
-- [1차]: 결과 집합의 상위 5개 행만 제한하여 반환합니다.
-- [2차]: 줄 세운 메뉴들 중에서 가장 비싼 5개만 골라 VIP 추천판에 올립니다.
LIMIT 5;

-- ------------------------------------------------------------------------------
-- Q02. VIP 이상 등급 회원의 이름, 이메일, 포인트 및 가입일 조회
-- [쿼리 목적]: 로열티 프로그램 대상자(VIP/VVIP)를 필터링하여 맞춤형 감사 프로모션 발송 명단 확보
-- ------------------------------------------------------------------------------
-- [1차]: 회원 ID, 이름, 이메일, 등급, 포인트 컬럼을 선택합니다.
-- [2차]: 우수 고객 감사 선물을 보내기 위해 고객 수첩에서 이름, 연락처, 적립 포인트를 확인합니다.
SELECT member_id, name, email, grade, points
-- [1차]: members 테이블을 조회 대상으로 지정합니다.
-- [2차]: 전체 고객 명단이 적힌 회원 명부 책장을 엽니다.
FROM members
-- [1차]: 회원 등급이 'VIP' 또는 'VVIP' 집합에 포함되는 조건만 필터링합니다.
-- [2차]: 명찰에 VIP나 VVIP 도장이 쾅 찍혀 있는 특별 고객들만 골라냅니다.
WHERE grade IN ('VIP', 'VVIP')
-- [1차]: 보유 포인트를 기준으로 가장 많은 순서대로 내림차순 정렬합니다.
-- [2차]: 스탬프 포인트를 가장 많이 모아둔 진정한 카페 찐팬 순서대로 줄을 세웁니다.
ORDER BY points DESC;

-- ------------------------------------------------------------------------------
-- Q03. 결제 완료된 주문 중 15,000원 이상 고액 결제 건 조회
-- [쿼리 목적]: 단체 주문이나 선물용 고액 구매 건을 선별하여 매출 기여도가 높은 주문 패턴 분석
-- ------------------------------------------------------------------------------
-- [1차]: 주문 ID, 회원 ID, 결제 상태, 결제 금액, 결제 수단, 주문 일시를 선택합니다.
-- [2차]: 매출 분석을 위해 영수증에서 영수증 번호, 결제 금액, 결제 수단 정보를 확인합니다.
SELECT order_id, member_id, order_status, total_amount, payment_method, ordered_at
-- [1차]: orders 테이블에서 조회합니다.
-- [2차]: 카운터 포스기에서 출력된 결제 영수증 보관함을 엽니다.
FROM orders
-- [1차]: 결제 상태가 'COMPLETED' 또는 'PAID'이고 총액이 15,000원 이상인 복합 조건을 검사합니다.
-- [2차]: 주문이 정상적으로 완료되었으면서 영수증 금액이 15,000원을 넘는 묵직한 고액 영수증만 뽑아냅니다.
WHERE order_status IN ('COMPLETED', 'PAID')
  AND total_amount >= 15000
-- [1차]: 주문 일시 기준으로 가장 최근 주문부터 내림차순 정렬합니다.
-- [2차]: 가장 최근에 발생한 따끈따끈한 고액 결제 영수증부터 위로 올려놓습니다.
ORDER BY ordered_at DESC;

-- ------------------------------------------------------------------------------
-- Q04. 신용카드(CARD)로 결제된 주문을 최신순으로 3건씩 페이징(Paging) 조회
-- [쿼리 목적]: 결제 단말기 카드 전표 대조를 위해 최근 카드 매출 3건을 1페이지 단위로 조회
-- ------------------------------------------------------------------------------
-- [1차]: 주문 ID, 회원 ID, 결제 금액, 주문 일시를 선택합니다.
-- [2차]: 카드사 승인 내역과 맞추기 위해 영수증 번호와 결제 시각, 금액을 확인합니다.
SELECT order_id, member_id, total_amount, ordered_at
-- [1차]: orders 테이블을 조회합니다.
-- [2차]: 주문 영수증 파일함을 엽니다.
FROM orders
-- [1차]: 결제 수단이 'CARD'인 주문만 필터링합니다.
-- [2차]: 현금이나 카카오페이를 제외하고 오직 실물 신용카드로 긁은 영수증만 추립니다.
WHERE payment_method = 'CARD'
-- [1차]: 최신 주문 시각을 기준으로 내림차순 정렬합니다.
-- [2차]: 방금 긁은 최신 카드 결제 전표부터 맨 위에 오도록 정렬합니다.
ORDER BY ordered_at DESC
-- [1차]: 한 페이지에 3개 행만 표시하도록 제한(LIMIT)하고 첫 번째 페이지(OFFSET 0)를 지정합니다.
-- [2차]: 모바일 화면 한 페이지에 영수증을 딱 3장씩만 보여주기 위해 첫 장부터 3장만 끊어서 가져옵니다.
LIMIT 3 OFFSET 0;

-- ==============================================================================
-- [범주 2. 테이블 조인(JOIN) 쿼리 (4개)]
-- ==============================================================================

-- ------------------------------------------------------------------------------
-- Q05. [INNER JOIN] 주문 내역과 주문자의 회원 정보(이름, 등급, 전화번호) 매칭 조회
-- [쿼리 목적]: 각 영수증이 어떤 회원의 주문인지 주문자와 결제 정보를 한눈에 결합 확인
-- ------------------------------------------------------------------------------
-- [1차]: 주문 ID, 고객 이름, 등급, 전화번호, 결제 금액, 결제 상태를 선택합니다.
-- [2차]: 영수증 번호 옆에 손님의 실명과 고객 등급 명찰을 나란히 붙여서 확인합니다.
SELECT o.order_id, m.name AS member_name, m.grade, m.phone, o.total_amount, o.order_status
-- [1차]: 주문 테이블 orders를 기준(별칭 o)으로 설정합니다.
-- [2차]: 결제 영수증 더미(o)를 책상 왼쪽에 올려놓습니다.
FROM orders o
-- [1차]: members 테이블(별칭 m)과 member_id가 일치하는 행끼리 내부 조인(INNER JOIN)합니다.
-- [2차]: 영수증에 적힌 회원번호를 보고, 오른쪽 회원 수첩(m)에서 동일한 번호의 손님 인적사항을 찾아 풀로 딱 붙입니다.
INNER JOIN members m ON o.member_id = m.member_id
-- [1차]: 주문 ID 오름차순으로 정렬합니다.
-- [2차]: 1번 영수증부터 차례대로 가지런히 정리합니다.
ORDER BY o.order_id ASC;

-- ------------------------------------------------------------------------------
-- Q06. [INNER JOIN - 3개 테이블 다중 조인] 주문 상세 내역과 메뉴명, 카테고리명 결합 조회
-- [쿼리 목적]: 주방 바리스타가 주문서를 보고 어떤 카테고리의 어떤 메뉴를 제조해야 하는지 상세 확인
-- ------------------------------------------------------------------------------
-- [1차]: 주문상세 ID, 주문 ID, 카테고리명, 메뉴명, 수량, 판매단가, 계산된 품목 총액을 선택합니다.
-- [2차]: 바리스타 주문서에 '주문번호, 음료 종류, 메뉴명, 잔 수, 단가, 총금액'을 일목요연하게 표시합니다.
SELECT 
    oi.order_item_id,
    oi.order_id,
    c.category_name,
    m.menu_name,
    oi.quantity,
    oi.unit_price,
    (oi.quantity * oi.unit_price) AS line_total
-- [1차]: 주문 품목 테이블(order_items, 별칭 oi)을 주 테이블로 지정합니다.
-- [2차]: 영수증 속 세부 품목 한 줄 한 줄(oi)을 펼칩니다.
FROM order_items oi
-- [1차]: 메뉴 테이블(menus, 별칭 m)과 menu_id로 조인하여 메뉴 이름을 가져옵니다.
-- [2차]: 품목에 적힌 메뉴 바코드 번호를 메뉴판(m)과 대조해 '아메리카노', '초코라떼' 같은 메뉴 이름을 알아냅니다.
INNER JOIN menus m ON oi.menu_id = m.menu_id
-- [1차]: 카테고리 테이블(categories, 별칭 c)과 category_id로 조인하여 대분류를 가져옵니다.
-- [2차]: 그 메뉴가 '커피 서랍(c)'에 있는지 '디저트 서랍(c)'에 있는지 분류 꼬리표를 확인합니다.
INNER JOIN categories c ON m.category_id = c.category_id
-- [1차]: 주문 ID 오름차순, 주문 품목 ID 오름차순으로 정렬합니다.
-- [2차]: 주문 영수증 순서대로, 그리고 그 영수증 안의 첫 번째 줄부터 순서대로 나열합니다.
ORDER BY oi.order_id ASC, oi.order_item_id ASC;

-- ------------------------------------------------------------------------------
-- Q07. [LEFT JOIN] 회원 테이블 기준 주문 이력 조회 (한 번도 주문 안 한 신규/휴면 회원 발굴)
-- [쿼리 목적]: 회원가입만 하고 아직 한 번도 구매하지 않은 고객을 찾아 웰컴 할인 쿠폰 발송
-- ------------------------------------------------------------------------------
-- [1차]: 회원 ID, 이름, 이메일, 가입일, 주문 ID를 선택합니다.
-- [2차]: 전체 고객 명부에서 이름과 가입일을 확인하고, 주문 영수증이 붙어 있는지 확인합니다.
SELECT m.member_id, m.name, m.email, m.created_at, o.order_id
-- [1차]: members 테이블을 왼쪽 기준 테이블로 지정합니다.
-- [2차]: 카페에 가입한 모든 고객(m)을 빠짐없이 출석부에 세웁니다.
FROM members m
-- [1차]: orders 테이블과 LEFT JOIN하여 주문이 없는 회원도 누락 없이 NULL로 유지합니다.
-- [2차]: 각 손님 옆에 그 손님이 결제한 영수증(o)을 나란히 놓되, 주문한 적이 없는 손님 옆은 빈칸(NULL)으로 둡니다.
LEFT JOIN orders o ON m.member_id = o.member_id
-- [1차]: 오른쪽 주문 테이블의 order_id가 NULL인 행만 필터링합니다.
-- [2차]: 영수증 자리가 텅 빈(NULL) 손님, 즉 가입만 하고 커피를 한 번도 안 사 마신 신규/미구매 고객만 콕 집어냅니다.
WHERE o.order_id IS NULL;

-- ------------------------------------------------------------------------------
-- Q08. [LEFT JOIN + COALESCE] 전체 메뉴별 판매 집계 (주문 실적이 없는 비인기 메뉴도 0개로 표시)
-- [쿼리 목적]: 메뉴판의 모든 메뉴가 몇 개씩 팔렸는지 전수 조사하여 메뉴 개편/단종 여부 판단
-- ------------------------------------------------------------------------------
-- [1차]: 메뉴 ID, 메뉴명, 정가, NULL을 0으로 치환한 총 주문 수량을 선택합니다.
-- [2차]: 메뉴 이름과 가격, 그리고 여태까지 팔린 총 잔 수를 보여주되, 한 잔도 안 팔린 메뉴는 빈칸 대신 숫자 0으로 예쁘게 채웁니다.
SELECT 
    m.menu_id,
    m.menu_name,
    m.price,
    COALESCE(SUM(oi.quantity), 0) AS total_sold_quantity
-- [1차]: 전체 메뉴 테이블(menus, 별칭 m)을 왼쪽 기준 테이블로 지정합니다.
-- [2차]: 메뉴판에 적힌 모든 메뉴(m)를 하나도 빠짐없이 칠판에 적습니다.
FROM menus m
-- [1차]: 주문 품목 테이블(order_items, 별칭 oi)과 LEFT JOIN합니다.
-- [2차]: 손님들이 주문한 품목 영수증 조각(oi)들을 해당 메뉴 이름 밑에 모아 붙입니다.
LEFT JOIN order_items oi ON m.menu_id = oi.menu_id
-- [1차]: 메뉴 ID와 메뉴명으로 그룹화합니다.
-- [2차]: 같은 메뉴 이름별로 묶어 서랍 하나에 담습니다.
GROUP BY m.menu_id, m.menu_name, m.price
-- [1차]: 판매 수량이 높은 순서(내림차순)로 정렬합니다.
-- [2차]: 가장 불티나게 팔린 베스트셀러 메뉴부터 맨 위에 줄을 세웁니다.
ORDER BY total_sold_quantity DESC;

-- ------------------------------------------------------------------------------
-- Q09. [COUNT, AVG, MAX + GROUP BY] 카테고리별 등록 메뉴 수, 평균 가격, 최고 가격 집계
-- [쿼리 목적]: 카테고리별 가격대 분포와 상품 구색 현황을 파악하여 메뉴 가격 정책 수립
-- ------------------------------------------------------------------------------
-- [1차]: 카테고리 ID, 카테고리명, 메뉴 개수(COUNT), 평균 가격(AVG 반올림), 최고 가격(MAX)을 계산합니다.
-- [2차]: 각 코너별로 등록된 메뉴가 몇 개인지 세고, 평균 가격과 제일 비싼 메뉴 가격표를 뽑아냅니다.
SELECT 
    c.category_id,
    c.category_name,
    COUNT(m.menu_id) AS menu_count,
    ROUND(AVG(m.price), 0) AS avg_price,
    MAX(m.price) AS max_price
-- [1차]: 카테고리 테이블을 기준으로 메뉴 테이블과 INNER JOIN합니다.
-- [2차]: 카테고리 서랍(c)과 거기에 속한 메뉴 카드들(m)을 짝짓습니다.
FROM categories c
INNER JOIN menus m ON c.category_id = m.category_id
-- [1차]: 카테고리 ID와 카테고리 이름으로 그룹화합니다.
-- [2차]: '커피팀', '디저트팀', '차팀' 등으로 모둠을 만듭니다.
GROUP BY c.category_id, c.category_name
-- [1차]: 카테고리 ID 순으로 정렬합니다.
-- [2차]: 1번 카테고리부터 차례대로 깔끔하게 성적표를 출력합니다.
ORDER BY c.category_id ASC;

-- ------------------------------------------------------------------------------
-- Q10. [COUNT, SUM + GROUP BY] 고객별 총 결제 횟수 및 누적 구매 금액 집계
-- [쿼리 목적]: 고객별 LTV(고객 생애 가치)를 산출하여 충성 고객 순위 산정
-- ------------------------------------------------------------------------------
-- [1차]: 회원 ID, 이름, 등급, 총 주문 건수(COUNT), 누적 결제 총액(SUM)을 계산합니다.
-- [2차]: 손님마다 몇 번 결제했는지 횟수를 세고, 우리 카페에 쓴 총 결제 금액을 합산합니다.
SELECT 
    m.member_id,
    m.name AS member_name,
    m.grade,
    COUNT(o.order_id) AS order_count,
    SUM(o.total_amount) AS total_spent
-- [1차]: 회원 테이블과 주문 테이블을 내부 조인합니다.
-- [2차]: 결제 내역이 있는 손님(m)과 그 손님의 영수증들(o)을 묶습니다.
FROM members m
INNER JOIN orders o ON m.member_id = o.member_id
-- [1차]: 취소되지 않고 유효한 주문('COMPLETED', 'PAID')만 집계에 반영합니다.
-- [2차]: 도중에 결제 취소한 영수증은 매출에서 쏙 빼고 정상 영수증만 합칩니다.
WHERE o.order_status IN ('COMPLETED', 'PAID')
-- [1차]: 회원 단위로 그룹화합니다.
-- [2차]: 손님 이름별로 영수증을 모아 작은 더미를 만듭니다.
GROUP BY m.member_id, m.name, m.grade
-- [1차]: 총 결제 금액 기준 내림차순 정렬합니다.
-- [2차]: 우리 매장에 가장 많은 돈을 써주신 고마운 손님부터 1등, 2등 순서로 랭킹을 매깁니다.
ORDER BY total_spent DESC;

-- ------------------------------------------------------------------------------
-- Q11. [GROUP BY + HAVING] 2회 이상 주문한 단골 고객 및 총 주문 금액 필터링
-- [쿼리 목적]: 단순 일회성 방문 고객을 제외하고, 2회 이상 재방문한 찐단골만 선별
-- ------------------------------------------------------------------------------
-- [1차]: 회원 ID, 이름, 주문 횟수, 총 결제액을 선택합니다.
-- [2차]: 단골 전용 VIP 혜택을 드리기 위해 손님 이름과 방문 횟수를 확인합니다.
SELECT 
    m.member_id,
    m.name AS member_name,
    COUNT(o.order_id) AS repeat_orders,
    SUM(o.total_amount) AS total_spent
-- [1차]: members와 orders를 조인합니다.
-- [2차]: 손님 명부와 결제 영수증을 결합합니다.
FROM members m
INNER JOIN orders o ON m.member_id = o.member_id
-- [1차]: 결제 유효 건만 대상으로 합니다.
-- [2차]: 유효한 결제 영수증만 카운트합니다.
WHERE o.order_status IN ('COMPLETED', 'PAID')
-- [1차]: 회원별로 그룹화합니다.
-- [2차]: 손님별로 묶어 그룹 바구니를 만듭니다.
GROUP BY m.member_id, m.name
-- [1차]: 집계된 주문 건수가 2건 이상인 그룹만 HAVING 절로 통과시킵니다.
-- [2차]: 한 번만 방문하고 가신 분들은 거르고, 매장에 최소 2번 이상 오신 단골 바구니만 쏙 골라냅니다.
HAVING COUNT(o.order_id) >= 2
-- [1차]: 재방문 횟수 내림차순 정렬합니다.
-- [2차]: 가장 자주 찾아오신 손님부터 줄을 세웁니다.
ORDER BY repeat_orders DESC, total_spent DESC;

-- ------------------------------------------------------------------------------
-- Q12. [중첩 서브쿼리] 전체 메뉴의 평균 가격보다 비싼 고급 메뉴 목록 조회
-- [쿼리 목적]: 메뉴 전체 평균 단가를 자동 계산하고, 그보다 높은 고부가가치 메뉴 선별
-- ------------------------------------------------------------------------------
-- [1차]: 메뉴 ID, 메뉴명, 가격 컬럼을 프로젝션합니다.
-- [2차]: 프리미엄 메뉴 추천 리스트에 띄울 메뉴 번호와 이름, 가격을 가져옵니다.
SELECT menu_id, menu_name, price
-- [1차]: menus 테이블에서 조회합니다.
-- [2차]: 메뉴판 서랍을 엽니다.
FROM menus
-- [1차]: 가격이 서브쿼리로 계산된 전체 메뉴 평균 가격보다 큰 행만 비교 필터링합니다.
-- [2차]: 서브쿼리 계산기(괄호 안)가 먼저 전체 메뉴의 평균 가격(약 7,200원)을 구한 뒤, 그 기준선보다 비싼 고급 메뉴만 쏙 뽑아냅니다.
WHERE price > (
    -- [1차]: 전체 menus 테이블의 price 평균값을 단일 스칼라(Scalar) 값으로 반환하는 서브쿼리입니다.
    -- [2차]: 카페 전체 메뉴들의 평균 가격을 구하는 보조 계산기입니다.
    SELECT AVG(price) FROM menus
)
-- [1차]: 가격 내림차순으로 정렬합니다.
-- [2차]: 가장 비싼 프리미엄 메뉴부터 맨 위에 나열합니다.
ORDER BY price DESC;

-- ------------------------------------------------------------------------------
-- Q13. [UPDATE] 누적 결제액 30,000원 이상인 일반 회원을 VIP로 승급 및 포인트 2,000점 지급
-- [쿼리 목적]: 결제 실적 기반 회원 등급 자동 승급 및 리워드 포인트 지급 비즈니스 로직
-- ------------------------------------------------------------------------------
-- [1차]: members 테이블의 grade와 points 컬럼 값을 수정합니다.
-- [2차]: 단골 자격을 획득한 손님의 명찰을 'VIP'로 바꿔 달아드리고, 보너스 포인트 2,000점을 통장에 얹어 드립니다.
UPDATE members
SET grade = 'VIP',
    points = points + 2000
-- [1차]: 수정 대상이 되는 회원을 서브쿼리로 필터링합니다.
-- [2차]: 과거 유효 결제 총액이 30,000원을 돌파한 우수 고객님들의 회원번호를 찾아서 그분들만 승급시킵니다.
WHERE member_id IN (
    -- [1차]: 결제 완료된 주문의 총합이 30,000원 이상인 회원 ID를 도출하는 서브쿼리입니다.
    -- [2차]: 누적 결제액 3만 원 이상을 달성한 VIP 승급 대상자 명단을 추리는 집계 계산기입니다.
    SELECT o.member_id
    FROM orders o
    WHERE o.order_status IN ('COMPLETED', 'PAID')
    GROUP BY o.member_id
    HAVING SUM(o.total_amount) >= 30000
)
-- [1차]: 이미 VIP나 VVIP인 회원은 제외하고 NORMAL 등급만 승급 대상으로 지정합니다.
-- [2차]: 이미 최고 등급이신 분들은 건너뛰고, 일반(NORMAL) 등급이었던 분들만 VIP로 올려드립니다.
AND grade = 'NORMAL';

-- ------------------------------------------------------------------------------
-- Q14. [DELETE] 주문 상세 품목이 없거나 취소(CANCELLED)된 비정상 주문 레코드 정리
-- [쿼리 목적]: 주문 상태가 'CANCELLED'인 결제 취소 건을 안전하게 삭제하여 DB 공간 확보
-- ------------------------------------------------------------------------------
-- [1차]: orders 테이블에서 특정 조건의 레코드를 삭제합니다.
-- [2차]: 카운터 영수증 보관함에서 취소 도장이 찍힌 불필요한 영수증을 골라내어 파쇄기에 넣습니다.
DELETE FROM orders
-- [1차]: 주문 상태가 'CANCELLED'인 주문을 식별합니다.
-- [2차]: 영수증 상태가 '주문 취소(CANCELLED)'로 끝난 영수증만 휴지통으로 보냅니다. (자식 테이블 품목은 ON DELETE CASCADE로 자동 함께 파쇄됨)
WHERE order_status = 'CANCELLED';

-- ------------------------------------------------------------------------------
-- Q15. [CREATE INDEX + EXPLAIN QUERY PLAN]
-- [적용 이유]: 대규모 서비스에서 회원별/주문일시별 주문 내역 조회 쿼리가 가장 빈번하므로,
--             (member_id, ordered_at) 복합 인덱스를 생성하여 풀 테이블 스캔(Full Scan)을
--             방지하고 O(log N) 인덱스 스캔(Index Scan)으로 조회 속도를 비약적으로 단축합니다.
-- ------------------------------------------------------------------------------
-- [1차]: orders 테이블의 member_id와 ordered_at 컬럼을 결합한 복합 B-Tree 인덱스를 생성합니다.
-- [2차]: 도서관에서 책을 찾기 쉽게 '손님 회원번호와 주문 날짜'가 적힌 빠른 색인용 인덱스 견출지 스티커를 만듭니다.
CREATE INDEX IF NOT EXISTS idx_orders_member_ordered_at 
ON orders (member_id, ordered_at DESC);

-- [1차]: 특정 회원의 최근 주문을 조회할 때 인덱스가 실제로 활용되는지 실행 계획(EXPLAIN QUERY PLAN)으로 확인합니다.

    # --------------------------------------------------------------------------
    def test_cascade_delete_order_items(self):
        # [1차]: 1번 주문을 삭제했을 때, 자식 테이블(order_items)의 1번 주문 관련 품목들도 함께 자동 연쇄 삭제되는지 검증합니다.
        # [2차]: 1번 영수증을 파쇄기에 넣으면 영수증에 딸린 품목 내역들도 함께 깔끔히 분쇄되는지 확인합니다.
        self.cursor.execute("SELECT COUNT(*) FROM order_items WHERE order_id = 1;")
        initial_items = self.cursor.fetchone()[0]
        self.assertGreater(initial_items, 0)

        # [1차]: 1번 주문 삭제
        self.cursor.execute("DELETE FROM orders WHERE order_id = 1;")

        # [1차]: 자식 품목 수 조회 -> 0개 확인
        self.cursor.execute("SELECT COUNT(*) FROM order_items WHERE order_id = 1;")
        remaining_items = self.cursor.fetchone()[0]
        self.assertEqual(remaining_items, 0, "부모 주문 삭제 시 자식 품목이 연쇄 삭제(CASCADE)되지 않았습니다.")

    # 7. 핵심 쿼리 15개 무결성 전수 검증
    # --------------------------------------------------------------------------
    def test_q01_to_q15_execution_integrity(self):
        # [Q01: 기본 조회 1] 판매 중 메뉴 상위 5개 가격 내림차순
        self.cursor.execute("""
            SELECT menu_id, menu_name, price FROM menus
            WHERE is_available = 1 ORDER BY price DESC LIMIT 5;
        """)
        rows_q01 = self.cursor.fetchall()
        self.assertEqual(len(rows_q01), 5)
        self.assertGreaterEqual(rows_q01[0][2], rows_q01[1][2])  # 내림차순 확인

        # [Q02: 기본 조회 2] VIP 이상 회원 조회
        self.cursor.execute("""
            SELECT member_id, name, grade FROM members
            WHERE grade IN ('VIP', 'VVIP') ORDER BY points DESC;
        """)
        rows_q02 = self.cursor.fetchall()
        self.assertGreaterEqual(len(rows_q02), 3)

        # [Q03: 기본 조회 3] 15,000원 이상 고액 결제 건
        self.cursor.execute("""
            SELECT order_id, total_amount FROM orders
            WHERE order_status IN ('COMPLETED', 'PAID') AND total_amount >= 15000;
        """)
        rows_q03 = self.cursor.fetchall()
        self.assertGreaterEqual(len(rows_q03), 3)

        # [Q04: 기본 조회 4] CARD 결제 페이징 조회 (LIMIT 3 OFFSET 0)
        self.cursor.execute("""
            SELECT order_id FROM orders WHERE payment_method = 'CARD' ORDER BY ordered_at DESC LIMIT 3 OFFSET 0;
        """)
        rows_q04 = self.cursor.fetchall()
        self.assertEqual(len(rows_q04), 3)

        # [Q05: 조인 1 - INNER] 주문과 회원 정보 조인
        self.cursor.execute("""
            SELECT o.order_id, m.name FROM orders o
            INNER JOIN members m ON o.member_id = m.member_id;
        """)
        rows_q05 = self.cursor.fetchall()
        self.assertGreaterEqual(len(rows_q05), 10)

        # [Q06: 조인 2 - 3개 테이블 다중 조인] 주문상세 + 메뉴 + 카테고리
        self.cursor.execute("""
            SELECT oi.order_item_id, c.category_name, m.menu_name, (oi.quantity * oi.unit_price)
            FROM order_items oi
            INNER JOIN menus m ON oi.menu_id = m.menu_id
            INNER JOIN categories c ON m.category_id = c.category_id;
        """)
        rows_q06 = self.cursor.fetchall()
        self.assertGreaterEqual(len(rows_q06), 15)

        # [Q07: 조인 3 - LEFT JOIN] 미주문 회원 발굴
        self.cursor.execute("""
            SELECT m.member_id, m.name FROM members m
            LEFT JOIN orders o ON m.member_id = o.member_id
            WHERE o.order_id IS NULL;
        """)
        rows_q07 = self.cursor.fetchall()
        # 샘플 데이터에 주문이 없는 회원은 9번 장우진, 10번 한예원 2명
        self.assertEqual(len(rows_q07), 2)

        # [Q08: 조인 4 - LEFT JOIN + COALESCE] 전체 메뉴 판매량 집계
        self.cursor.execute("""
            SELECT m.menu_id, m.menu_name, COALESCE(SUM(oi.quantity), 0) AS total_sold
            FROM menus m
            LEFT JOIN order_items oi ON m.menu_id = oi.menu_id
            GROUP BY m.menu_id, m.menu_name, m.price
            ORDER BY total_sold DESC;
        """)
        rows_q08 = self.cursor.fetchall()
        self.assertEqual(len(rows_q08), 15)  # 총 메뉴 15개 모두 누락 없이 반환

        # [Q09: 집계 1 - COUNT, AVG, MAX + GROUP BY] 카테고리별 통계
        self.cursor.execute("""
            SELECT c.category_name, COUNT(m.menu_id), ROUND(AVG(m.price), 0), MAX(m.price)
            FROM categories c
            INNER JOIN menus m ON c.category_id = m.category_id
            GROUP BY c.category_id, c.category_name;
        """)
        rows_q09 = self.cursor.fetchall()
        self.assertGreaterEqual(len(rows_q09), 10)

        # [Q10: 집계 2 - COUNT, SUM + GROUP BY] 고객별 누적 구매액 및 주문수
        self.cursor.execute("""
            SELECT m.name, COUNT(o.order_id), SUM(o.total_amount)
            FROM members m
            INNER JOIN orders o ON m.member_id = o.member_id
            WHERE o.order_status IN ('COMPLETED', 'PAID')
            GROUP BY m.member_id, m.name;
        """)
        rows_q10 = self.cursor.fetchall()
        self.assertGreaterEqual(len(rows_q10), 6)

        # [Q11: 집계 3 - HAVING] 2회 이상 주문한 단골 필터링
        self.cursor.execute("""
            SELECT m.name, COUNT(o.order_id)
            FROM members m
            INNER JOIN orders o ON m.member_id = o.member_id
            WHERE o.order_status IN ('COMPLETED', 'PAID')
            GROUP BY m.member_id, m.name
            HAVING COUNT(o.order_id) >= 2;
        """)
        rows_q11 = self.cursor.fetchall()
        self.assertGreaterEqual(len(rows_q11), 3)

        # [Q12: 서브쿼리 1] 전체 메뉴 평균 가격보다 비싼 메뉴
        self.cursor.execute("""
            SELECT menu_name, price FROM menus
            WHERE price > (SELECT AVG(price) FROM menus);
        """)
        rows_q12 = self.cursor.fetchall()
        self.assertGreaterEqual(len(rows_q12), 3)

        # [Q13: 데이터 수정 - UPDATE] 3만원 이상 구매 일반 회원 VIP 승급
        self.cursor.execute("""
            UPDATE members SET grade = 'VIP', points = points + 2000
            WHERE member_id IN (
                SELECT o.member_id FROM orders o
                WHERE o.order_status IN ('COMPLETED', 'PAID')
                GROUP BY o.member_id HAVING SUM(o.total_amount) >= 30000
            ) AND grade = 'NORMAL';
        """)
        # 변경 행 수 검증
        self.assertGreaterEqual(self.cursor.rowcount, 0)

        # [Q14: 데이터 삭제 - DELETE] CANCELLED 주문 정리
        self.cursor.execute("SELECT COUNT(*) FROM orders WHERE order_status = 'CANCELLED';")
        cancelled_before = self.cursor.fetchone()[0]
        self.assertEqual(cancelled_before, 1)

        self.cursor.execute("DELETE FROM orders WHERE order_status = 'CANCELLED';")
        self.assertEqual(self.cursor.rowcount, 1)

        # [Q15: 인덱스 생성 및 실행 계획 확인]
        self.cursor.execute("""
            CREATE INDEX IF NOT EXISTS idx_orders_member_ordered_at 
            ON orders (member_id, ordered_at DESC);
        """)
        self.cursor.execute("""
            EXPLAIN QUERY PLAN
            SELECT order_id, total_amount, ordered_at FROM orders
            WHERE member_id = 1 ORDER BY ordered_at DESC;
        """)
        plan = self.cursor.fetchall()
        # 실행 계획 문자열에 idx_orders_member_ordered_at 인덱스가 사용되었는지 검증
        plan_str = " ".join([str(p) for p in plan])
        self.assertIn("idx_orders_member_ordered_at", plan_str)

def setup_in_memory_db(root_dir: Path) -> sqlite3.Connection:
    # [1차]: 인메모리 SQLite 연결을 생성하고 스키마와 샘플 데이터를 로드합니다.
    # [2차]: 평가자가 기다릴 필요 없이 RAM 메모리에서 즉시 0.01초 만에 테스트용 카페 DB를 가동합니다.
    conn = sqlite3.connect(":memory:")
    cursor = conn.cursor()
    cursor.execute("PRAGMA foreign_keys = ON;")

    schema_file = root_dir / "sql" / "01_schema.sql"
    with open(schema_file, "r", encoding="utf-8") as f:
        cursor.executescript(f.read())

    data_file = root_dir / "sql" / "02_sample_data.sql"
    with open(data_file, "r", encoding="utf-8") as f:
        cursor.executescript(f.read())

def format_table(headers: list, rows: list) -> str:
    # [1차]: 결과 집합을 깔끔한 아스키 테이블 문자열로 포맷팅합니다.
    # [2차]: 표 형태로 데이터를 예쁘게 줄 맞춰서 콘솔에 보여주는 자동 줄맞춤 자입니다.
    if not rows:
        return "(조회된 결과가 없습니다 / 0 rows)"

    col_widths = [len(str(h)) for h in headers]
    for row in rows:
        for i, val in enumerate(row):
            col_widths[i] = max(col_widths[i], len(str(val)))

    header_str = " | ".join(f"{str(headers[i]).ljust(col_widths[i])}" for i in range(len(headers)))
    sep_str = "-+-".join("-" * col_widths[i] for i in range(len(headers)))
    
    row_lines = []
    for row in rows:
        row_lines.append(" | ".join(f"{str(row[i]).ljust(col_widths[i])}" for i in range(len(headers))))

    return f"{header_str}\n{sep_str}\n" + "\n".join(row_lines)


def format_markdown_table(headers: list, rows: list) -> str:
    # [1차]: 마크다운 문서용 테이블 문자열을 생성합니다.
    # [2차]: 마크다운 문서에 바로 붙여넣을 수 있는 표를 만듭니다.
    if not rows:
        return "*결과 없음 (0 rows)*"

    header_str = "| " + " | ".join(headers) + " |"
    sep_str = "| " + " | ".join(["---"] * len(headers)) + " |"
    row_strs = ["| " + " | ".join(str(cell) for cell in row) + " |" for row in rows]
    return f"{header_str}\n{sep_str}\n" + "\n".join(row_strs)

def create_erd_diagram(output_path: Path):

    fig.patch.set_facecolor("#FFFFFF")

            "UK: category_name (TEXT)",

            "price (INT)",
            "is_available (INT)"
        ]),
        ("members (고객 회원)", 0.5, 1.0, 3.2, 2.7, "#FFF3E0", "#F57C00", [
            "PK: member_id (INT)",
            "UK: email (TEXT)",

            "PK: order_id (INT)",
            "FK: member_id (INT)",
            "order_status (TEXT)",

            "payment_method (TEXT)"
        ]),

            "PK: order_item_id (INT)",
            "FK: order_id (INT)",
            "FK: menu_id (INT)",
            "quantity (INT)",
            "unit_price (INT)"
        ])
    ]

    for name, x, y, w, h, bg_color, border_color, cols in tables:
        rect = patches.FancyBboxPatch((x, y), w, h, boxstyle="round,pad=0.1",
                                      facecolor=bg_color, edgecolor=border_color, linewidth=2)
        ax.add_patch(rect)
        # 테이블 헤더
        ax.text(x + w / 2, y + h - 0.35, name, ha="center", va="center", fontsize=11, fontweight="bold", color=border_color)

    ax.text(8.7, 3.2, "1 : N", ha="center", fontsize=9, fontweight="bold", color="#7B1FA2")

    # 4. menus -> order_items (1:N)

                arrowprops=dict(arrowstyle="->", color="#388E3C", lw=2, mutation_scale=15))

    fig.patch.set_facecolor("#FFFFFF")

    # 부모 테이블 (members)
    rect1 = patches.FancyBboxPatch((0.8, 1.5), 4.2, 3.5, boxstyle="round,pad=0.1",
                                  facecolor="#E8EAF6", edgecolor="#3F51B5", linewidth=2)
    ax.add_patch(rect1)
    ax.text(2.9, 4.6, "[부모 1] members (회원 테이블)", ha="center", fontsize=11, fontweight="bold", color="#3F51B5")
    parent_data = [
        "member_id (PK) | 이름   | 등급",
        "------------------------------",
        "  1 (김민준)    | 김민준 | VVIP",
        "  2 (이지은)    | 이지은 | VIP ",
        "  3 (박서준)    | 박서준 | NORMAL"
    ]
    y_pos = 4.0
    for line in parent_data:
        ax.text(1.1, y_pos, line, fontsize=9.5)
        y_pos -= 0.5

    # 자식 테이블 (orders)
    rect2 = patches.FancyBboxPatch((6.8, 1.0), 4.5, 4.2, boxstyle="round,pad=0.1",
                                  facecolor="#FBE9E7", edgecolor="#D84315", linewidth=2)
    ax.add_patch(rect2)
    ax.text(9.05, 4.8, "[자식 N] orders (주문 테이블)", ha="center", fontsize=11, fontweight="bold", color="#D84315")
    child_data = [
        "order_id (PK) | member_id (FK) | 금액",
        "--------------------------------------",
        "  101         | 1 (김민준 참조) | 13,800",
        "  102         | 2 (이지은 참조) | 12,300",
        "  103         | 1 (김민준 참조) |  5,000",
        "  104         | 1 (김민준 참조) | 16,000",
        "  105         | 3 (박서준 참조) |  4,500"
    ]
    y_pos = 4.3
    for line in child_data:
        ax.text(7.0, y_pos, line, fontsize=9)
        y_pos -= 0.45

    # 참조 화살표 연결
    ax.annotate("", xy=(7.0, 3.85), xytext=(5.0, 3.0),
                arrowprops=dict(arrowstyle="->", color="#D84315", lw=2, mutation_scale=15))
    ax.text(6.0, 3.6, "김민준 1:N 복수 주문", ha="center", fontsize=9, fontweight="bold", color="#D84315")

    # 핵심 해설 텍스트 박스
    rect_box = patches.FancyBboxPatch((1.5, 0.2), 9.0, 0.6, boxstyle="square,pad=0.05",
                                     facecolor="#ECEFF1", edgecolor="#90A4AE", linewidth=1)
    ax.add_patch(rect_box)
    ax.text(6.0, 0.5, "핵심 규칙: 1:N 관계에서 외래키(FK)는 무조건 'N(다)' 쪽 테이블에 위치해야 중복 없이 연결됩니다!",
            ha="center", va="center", fontsize=9.5, fontweight="bold", color="#37474F")

    ax.set_xlim(0, 12)
    ax.set_ylim(0, 5.5)
    ax.axis("off")
    ax.set_title("1:N (일대다) 관계 및 외래키(FK) 참조 원리", fontsize=15, fontweight="bold", pad=15)

    plt.tight_layout()
    plt.savefig(output_path, bbox_inches="tight", dpi=300)
    plt.close()
    print(f"[생성 완료] {output_path}")


def create_join_diagram(output_path: Path):
    # [1차]: INNER JOIN과 LEFT JOIN의 결과 집합 차이를 명확히 시각화하는 다이어그램을 생성합니다.
    # [2차]: 교집합만 건지는 INNER JOIN과, 왼쪽 손님을 다 살리는 LEFT JOIN의 차이를 그림으로 보여줍니다.
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(13, 6), dpi=300)
    fig.patch.set_facecolor("#FFFFFF")

    # [좌측] INNER JOIN (교집합)
    ax1.set_facecolor("#F9FBE7")
    circle1 = patches.Circle((2.5, 3.0), 1.8, facecolor="#81C784", alpha=0.5, edgecolor="#2E7D32", lw=2)
    circle2 = patches.Circle((4.5, 3.0), 1.8, facecolor="#64B5F6", alpha=0.5, edgecolor="#1565C0", lw=2)
    ax1.add_patch(circle1)
    ax1.add_patch(circle2)
    ax1.text(1.8, 3.0, "회원 명부\n(members)", ha="center", va="center", fontsize=10, fontweight="bold")
    ax1.text(5.2, 3.0, "주문 내역\n(orders)", ha="center", va="center", fontsize=10, fontweight="bold")
    ax1.text(3.5, 3.0, "★ 양쪽 모두\n존재하는\n구매 회원\n(INNER)", ha="center", va="center", fontsize=10, fontweight="bold", color="#B71C1C")
    ax1.text(3.5, 0.8, "주문 이력이 있는 활성 고객만 조회\n(미구매 신규 회원은 탈락)", ha="center", fontsize=9.5, color="#333333")
    ax1.set_xlim(0, 7)
    ax1.set_ylim(0, 5.5)
    ax1.axis("off")
    ax1.set_title("[1] INNER JOIN (내부 조인 - 교집합)", fontsize=13, fontweight="bold", pad=10)

    # [우측] LEFT JOIN (왼쪽 기준 + 빈자리 NULL)
    ax2.set_facecolor("#FFFDE7")
    circle3 = patches.Circle((2.5, 3.0), 1.8, facecolor="#FFB74D", alpha=0.8, edgecolor="#E65100", lw=2)
    circle4 = patches.Circle((4.5, 3.0), 1.8, facecolor="#E0E0E0", alpha=0.4, edgecolor="#757575", lw=1.5, ls="--")
    ax2.add_patch(circle3)
    ax2.add_patch(circle4)
    ax2.text(1.8, 3.0, "전체 회원\n(누락 없음)", ha="center", va="center", fontsize=10, fontweight="bold", color="#BF360C")
    ax2.text(5.2, 3.0, "주문 내역\n(orders)", ha="center", va="center", fontsize=9.5, color="#757575")
    ax2.text(3.5, 3.0, "구매 이력\n매칭", ha="center", va="center", fontsize=9.5, fontweight="bold")
    ax2.text(3.5, 0.8, "주문 없는 신규 회원도 NULL로 모두 보존!\n(WHERE orders.id IS NULL로 미구매자 발굴)", ha="center", fontsize=9.5, color="#333333")
    ax2.set_xlim(0, 7)
    ax2.set_ylim(0, 5.5)
    ax2.axis("off")
    ax2.set_title("[2] LEFT JOIN (외부 조인 - 왼쪽 전체 보존)", fontsize=13, fontweight="bold", pad=10)

    plt.tight_layout()
    plt.savefig(output_path, bbox_inches="tight", dpi=300)
    plt.close()
    print(f"[생성 완료] {output_path}")


def create_pipeline_diagram(output_path: Path):
    # [1차]: SQL 구문의 물리적 실제 실행 순서(파이프라인)를 보여주는 플로우차트를 생성합니다.
    # [2차]: SQL이 머릿속에서 어떤 순서로 데이터를 요리해 나가는지 요리 레시피 순서도를 그립니다.
    fig, ax = plt.subplots(figsize=(13, 5), dpi=300)
    ax.set_facecolor("#F5F5F5")
    fig.patch.set_facecolor("#FFFFFF")

    steps = [
        ("1. FROM / JOIN", "조회 대상 테이블 준비 & 결합", "#E1F5FE", "#0288D1"),
        ("2. WHERE", "행 단위 개별 조건 필터링", "#E8F5E9", "#2E7D32"),
        ("3. GROUP BY", "공통 컬럼 기준 그룹 묶기", "#FFF8E1", "#F57F17"),
        ("4. HAVING", "집계 결과 그룹 조건 필터링", "#FBE9E7", "#D84315"),
        ("5. SELECT", "최종 출력 컬럼 프로젝션", "#EDE7F6", "#512DA8"),
        ("6. ORDER BY", "지정 기준 오름/내림차순 정렬", "#F3E5F5", "#8E24AA"),
        ("7. LIMIT", "최종 출력 행 개수 제한", "#ECEFF1", "#37474F")
    ]

    x_start = 0.5
    box_w = 1.45
    box_h = 2.2

    for i, (title, desc, bg, border) in enumerate(steps):
        rect = patches.FancyBboxPatch((x_start, 1.2), box_w, box_h, boxstyle="round,pad=0.08",
                                      facecolor=bg, edgecolor=border, linewidth=2)
        ax.add_patch(rect)
        ax.text(x_start + box_w / 2, 2.7, title, ha="center", va="center", fontsize=10, fontweight="bold", color=border)
        ax.text(x_start + box_w / 2, 1.9, desc, ha="center", va="center", fontsize=8.5, color="#333333", wrap=True)

        if i < len(steps) - 1:
            ax.annotate("", xy=(x_start + box_w + 0.3, 2.3), xytext=(x_start + box_w + 0.05, 2.3),
                        arrowprops=dict(arrowstyle="->", color="#455A64", lw=2, mutation_scale=12))

        x_start += box_w + 0.32

    ax.set_xlim(0, 13.0)
    ax.set_ylim(0.5, 4.2)
    ax.axis("off")
    ax.set_title("SQL 엔진의 실제 내부 쿼리 실행 파이프라인 (Execution Flow)", fontsize=15, fontweight="bold", pad=15)

    plt.tight_layout()
    plt.savefig(output_path, bbox_inches="tight", dpi=300)
    plt.close()
    print(f"[생성 완료] {output_path}")
